Supabase SQL Schema setup

Jalankan query ini di SQL Editor Dashboard Supabase Anda jika tabel `chat_messages` belum dibuat:

CREATE TABLE IF NOT EXISTS public.chat_messages (
    id BIGSERIAL PRIMARY KEY,
    sender_role VARCHAR(20) NOT NULL,
    sender_name VARCHAR(100) NOT NULL,
    message TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Buka akses Realtime & RLS
ALTER TABLE public.chat_messages ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public access" ON public.chat_messages FOR ALL USING (true);
ALTER PUBLICATION supabase_realtime ADD TABLE public.chat_messages;
